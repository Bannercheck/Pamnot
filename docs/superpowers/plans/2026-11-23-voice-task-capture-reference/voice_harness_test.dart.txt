import 'dart:async';
import 'dart:convert';

import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:intl/date_symbol_data_local.dart';
import 'package:pamnot/core/quick_parse.dart';
import 'package:pamnot/data/store.dart';
import 'package:pamnot/models/settings.dart';
import 'package:pamnot/services/reminder_scheduler.dart';
import 'package:pamnot/services/speech_engine.dart';
import 'package:pamnot/state/app_controller.dart';
import 'package:pamnot/state/app_scope.dart';

import 'support/fake_speech_engine.dart';

/// Throwaway probe standing in for the real sheet: it only proves that a fake
/// engine + injected clock + tester.pump(duration) can drive the whole loop.
class _Probe extends StatefulWidget {
  const _Probe({required this.engine});
  final FakeSpeechEngine engine;
  @override
  State<_Probe> createState() => _ProbeState();
}

class _ProbeState extends State<_Probe> {
  StreamSubscription<SpeechEvent>? _sub;
  Timer? _gap, _hold, _idle;
  String _text = '';
  String _status = 'idle';

  @override
  void initState() {
    super.initState();
    _sub = widget.engine.events.listen(_on);
    _arm();
  }

  Future<void> _arm() async {
    if (await widget.engine.prepare('tr-TR') != SpeechAvailability.ready) {
      setState(() => _status = 'denied');
      return;
    }
    await widget.engine.start('tr-TR');
    _idle = Timer(const Duration(seconds: 6), () => setState(() => _status = 'closed'));
  }

  void _on(SpeechEvent e) {
    switch (e) {
      case SpeechPartial(:final text):
        _idle?.cancel();
        _gap?.cancel();
        _gap = Timer(const Duration(milliseconds: 1500), () => widget.engine.stop());
        setState(() {
          _text = text;
          _status = 'listening';
        });
      case SpeechFinal(:final text):
        final app = AppScope.of(context);
        final r = quickParse(text, app.now());
        setState(() {
          _status = 'preview:${r.title}';
        });
        _hold = Timer(const Duration(seconds: 2), () {
          app.addQuickTask(text);
          setState(() => _status = 'saved');
          _arm();
        });
      case SpeechFailure():
        setState(() => _status = 'error');
      case SpeechStarted() || SpeechLevel() || SpeechStopped():
        break;
    }
  }

  @override
  void dispose() {
    for (final t in [_gap, _hold, _idle]) {
      t?.cancel();
    }
    _sub?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => Column(children: [Text(_status), Text(_text)]);
}

void main() {
  setUpAll(() => initializeDateFormatting());
  final now = DateTime(2026, 10, 27, 9, 0);

  testWidgets('fake engine drives listen -> preview -> save -> re-arm -> idle close', (tester) async {
    final scheduler = NoopScheduler();
    final c = AppController(
      store: MemoryStore({'settings': jsonEncode(const AppSettings(language: AppLanguage.tr, seeded: true).toJson())}),
      clock: () => now,
      scheduler: scheduler,
      saveDelay: Duration.zero,
    );
    await tester.runAsync(c.load);
    final engine = FakeSpeechEngine();
    await tester.pumpWidget(AppScope(controller: c, child: MaterialApp(home: Scaffold(body: _Probe(engine: engine)))));
    await tester.pump();
    expect(engine.calls, ['prepare:tr-TR', 'start:tr-TR']);

    engine.say('yarın 10:00 toplantı');
    await tester.pump();
    expect(find.text('listening'), findsOneWidget);

    await tester.pump(const Duration(milliseconds: 1500)); // utterance gap -> stop -> final
    expect(find.text('preview:Toplantı'), findsOneWidget);

    await tester.pump(const Duration(seconds: 2)); // preview hold -> save -> re-arm
    expect(find.text('saved'), findsOneWidget);
    final t = c.tasks.single;
    expect(t.title, 'Toplantı');
    expect(t.due, DateTime(2026, 10, 28, 10));
    expect(t.hasTime, isTrue);
    // Today's quick-add never sets a reminder, so no task alert is produced.
    expect(t.remindAt, isNull);
    expect(c.alerts().where((a) => a.key.startsWith('t:')), isEmpty);

    await tester.pump(const Duration(seconds: 6)); // silence -> close
    expect(find.text('closed'), findsOneWidget);

    await tester.pumpWidget(const SizedBox());
    await tester.pump(const Duration(seconds: 3));
    c.dispose();
  });
}
